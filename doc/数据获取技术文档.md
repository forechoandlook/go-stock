## 1. 数据来源概述

Go-Stock 项目从多个渠道获取股票相关数据，包括实时行情、市场新闻、基本面数据等。以下是主要数据来源：

| 数据类型       | 来源        | API 地址                              | 更新频率  |
| -------------- | ----------- | ------------------------------------- | --------- |
| 股票实时数据   | 新浪财经    | http://hq.sinajs.cn                   | 实时      |
| 股票实时数据   | 腾讯财经    | http://qt.gtimg.cn                    | 实时      |
| 股票基本面数据 | Tushare     | http://api.tushare.pro                | 每日/每周 |
| 市场新闻       | 财联社电报  | https://www.cls.cn                    | 实时      |
| 市场新闻       | 新浪财经    | https://finance.sina.com.cn           | 实时      |
| 市场新闻       | TradingView | https://news-mediator.tradingview.com | 实时      |

## 3. 数据获取实现

### 3.1 股票实时数据获取

**文件：** `backend/data/stock_data_api.go`

#### 3.1.1 实现逻辑

1. **数据源选择**：根据股票代码类型选择不同的数据源

   - A股：新浪财经或腾讯财经
   - 港股：腾讯财经
   - 美股：新浪财经
2. **API 调用**：

   - 构建请求 URL，包含股票代码
   - 设置请求头（User-Agent、Referer等）
   - 发送 HTTP 请求获取数据
3. **数据解析**：

   - 解析不同格式的响应数据
   - 转换为统一的 StockInfo 结构
4. **数据存储**：

   - 检查数据库中是否存在该股票数据
   - 不存在则创建，存在则更新

#### 3.1.2 核心代码

```go
func (receiver StockDataApi) GetStockCodeRealTimeData(StockCodes ...string) (*[]StockInfo, error) {
	stockInfos := make([]StockInfo, 0)

	// 处理港股和A股代码
	hkcodes := slice.Filter(StockCodes, func(i int, s string) bool {
		return strutil.HasPrefixAny(s, []string{"hk", "HK", "sh", "sz"})
	})

	if hkcodes != nil && len(hkcodes) > 0 {
		// 构建腾讯财经 API URL
		hkcodesStr := slice.JoinFunc(hkcodes, ",", func(s string) string {
			if strutil.HasPrefixAny(s, []string{"hk", "HK"}) {
				return "r_" + strings.ToLower(s)
			} else {
				return strings.ToLower(s)
			}
		})
		url := fmt.Sprintf(txStockUrl, time.Now().Unix(), hkcodesStr)
	
		// 发送请求并解析数据
		resp, err := receiver.client.R().
			SetHeader("Host", "qt.gtimg.cn").
			SetHeader("Referer", "https://gu.qq.com/").
			Get(url)
	
		// 解析响应数据
		str := GB18030ToUTF8(resp.Body())
		dataStr := strutil.SplitAndTrim(strings.Trim(str, "\n"), ";")

		for _, data := range dataStr {
			stockData, err := ParseTxStockData(data)
			if err != nil {
				logger.SugaredLogger.Error(err.Error())
				continue
			}
			stockInfos = append(stockInfos, *stockData)
		
			// 异步存储数据
			go func() {
				var count int64
				db.Dao.Model(&StockInfo{}).Where("code = ?", stockData.Code).Count(&count)

				} else {
					db.Dao.Model(&StockInfo{}).Where("code = ?", stockData.Code).Updates(stockData)

		}
	}

	// 处理美股代码

		return !strutil.HasPrefixAny(s, []string{"hk", "HK", "sh", "sz"})
	})

	// 类似处理美股数据...

	return &stockInfos, err
}
```

```go
func (receiver StockDataApi) GetStockBaseInfo() {
	res := &TushareStockBasicResponse{}
	fields := "ts_code,symbol,name,area,industry,cnspell,market,list_date,act_name,act_ent_type,fullname,exchange,list_status,curr_type,enname,delist_date,is_hs"
	resp, err := receiver.client.R().
		SetHeader("content-type", "application/json").
		SetBody(&TushareRequest{
			ApiName: "stock_basic",
			Token:   receiver.config.TushareToken,
			Params:  nil,
			Fields:  fields,
		}).
		SetResult(res).
		Post(tushareApiUrl)

	// 保存响应到文件
	ioutil.WriteFile("stock_basic.json", resp.Body(), 0666)

	if err != nil {
		logger.SugaredLogger.Error(err.Error())
		return
	}
	if res.Code != 0 {
		logger.SugaredLogger.Error(res.Msg)
		return
	}

	// 解析并存储数据
	for _, item := range res.Data.Items {
		stock := &StockBasic{}
		data := map[string]any{}
		for _, field := range strings.Split(fields, ",") {
			idx := slice.IndexOf(res.Data.Fields, field)
			if idx == -1 {
				continue
			}
			data[field] = item[idx]
		}
		jsonData, _ := json.Marshal(data)
		err := json.Unmarshal(jsonData, stock)
		if err != nil {
			continue
		}
		stock.ID = 0
		db.Dao.Model(&StockBasic{}).FirstOrCreate(stock, &StockBasic{TsCode: stock.TsCode}).Where("ts_code = ?", stock.TsCode).Updates(stock)
	}
}
```

```go
func ParseTxStockData(data string) (*StockInfo, error) {
	datas := strutil.SplitAndTrim(data, "=", "\"")
	if len(datas) < 2 {
		return nil, fmt.Errorf("invalid data format")
	}
	var result map[string]string
	var err error
	if strutil.ContainsAny(datas[0], []string{"v_r_hk", "v_hk", "v_sz", "v_sh"}) {
		result, err = ParseTxHKStockData(datas)
	}

	// 转换为 StockInfo 结构
	marshal, err := json.Marshal(result)
	if err != nil {
		logger.SugaredLogger.Errorf("json.Marshal error:%s", err.Error())
		return nil, err
	}
	stockInfo := &StockInfo{}
	err = json.Unmarshal(marshal, &stockInfo)
	if err != nil {
		logger.SugaredLogger.Errorf("json.Unmarshal error:%s", err.Error())
		return nil, err
	}

	return stockInfo, nil
}
```

```go
func (m MarketNewsApi) GetSinaNews(crawlTimeOut uint) *[]models.Telegraph {
	news := &[]models.Telegraph{}
	response, _ := resty.New().SetTimeout(time.Duration(crawlTimeOut)*time.Second).R().
		SetHeader("Referer", "https://finance.sina.com.cn").
		SetHeader("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/117.0.0.0 Safari/537.36 Edg/117.0.2045.60").
		Get("https://zhibo.sina.com.cn/api/zhibo/feed?callback=callback&page=1&page_size=20&zhibo_id=152&tag_id=0&dire=f&dpc=1&pagesize=20&id=4161089&type=0&_=" + strconv.FormatInt(time.Now().Unix(), 10))

	// 处理 JSONP 响应
	js := string(response.Body())
	js = strutil.ReplaceWithMap(js, map[string]string{
		"try{callback(":  "var data=",
		");}catch(e){};": ";",
	})

	// 使用 otto 执行 JavaScript 解析
	vm := otto.New()
	_, err := vm.Run(js)
	if err != nil {
		logger.SugaredLogger.Error(err)
	}
	vm.Run("var result = data.result;")
	vm.Run("var resultData = result.data;")
	vm.Run("var feed = resultData.feed;")
	vm.Run("var feedStr = JSON.stringify(feed);")

	value, _ := vm.Get("feedStr")
	feed := make(map[string]any)
	err = json.Unmarshal([]byte(value.String()), &feed)
	if err != nil {
		logger.SugaredLogger.Errorf("json.Unmarshal error:%v", err.Error())
	}

	// 解析新闻列表
	var telegraphs []models.Telegraph
	if feed["list"] != nil {
		for _, item := range feed["list"].([]any) {
			telegraph := models.Telegraph{Source: "新浪财经"}
			data := item.(map[string]any)
			telegraph.Content = data["rich_text"].(string)
			telegraph.Title = strutil.SubInBetween(data["rich_text"].(string), "【", "】")
			telegraph.Time = strings.Split(data["create_time"].(string), " ")[1]
			dataTime, _ := time.ParseInLocation("2006-01-02 15:04:05", data["create_time"].(string), time.Local)
			if &dataTime != nil {
				telegraph.DataTime = &dataTime
			}
		
			// 处理标签
			tags := data["tag"].([]any)
			telegraph.SubjectTags = lo.Map(tags, func(tagItem any, index int) string {
				name := tagItem.(map[string]any)["name"].(string)
				tag := &models.Tags{
					Name: name,
					Type: "sina_subject",
				}
				db.Dao.Model(tag).Where("name=? and type=?", name, "sina_subject").FirstOrCreate(&tag)
				return name
			})
		
			// 存储数据
			if telegraph.Content != "" {
				telegraph.SentimentResult = AnalyzeSentiment(telegraph.Content).Description
				cnt := int64(0)
				if telegraph.Title == "" {
					db.Dao.Model(telegraph).Where("content=?", telegraph.Content).Count(&cnt)
				} else {
					db.Dao.Model(telegraph).Where("title=?", telegraph.Title).Count(&cnt)
				}
				if cnt == 0 {
					db.Dao.Create(&telegraph)
					telegraphs = append(telegraphs, telegraph)
				}
			}
		}
		return &telegraphs
	}

	return news
}
```

## 7. 可能的解析失败原因及处理

### 7.1 解析失败原因

1. **API 响应格式变化**：

   - 数据源可能更改 API 格式
   - 响应字段名称或结构变化
2. **网络请求失败**：

   - 网络连接超时
   - API 服务不可用
   - 反爬机制限制
3. **数据结构不符合预期**：

   - 响应数据为空
   - 字段缺失
   - 数据类型错误
4. **编码问题**：

   - 字符编码转换错误
   - 特殊字符处理

### 7.2 错误处理机制

1. **网络错误处理**：

   - 设置合理的超时时间
   - 实现重试机制
   - 记录详细的错误日志
2. **解析错误处理**：

   - 严格的类型检查
   - 字段存在性检查
   - 错误恢复机制
3. **数据验证**：

   - 验证数据完整性
   - 过滤无效数据
   - 确保数据一致性

### 7.3 容错措施

1. **多数据源备份**：

   - 同一数据从多个来源获取
   - 当主数据源失败时切换到备用源
2. **缓存机制**：

   - 缓存历史数据
   - 在数据源失败时使用缓存数据
3. **优雅降级**：

   - 部分功能不可用时，确保其他功能正常
   - 提供合理的错误提示

### 12.2 添加新数据源

#### 12.2.1 创建数据 API 文件

在 `backend/data/` 目录下创建新的 API 文件（如 `new_data_api.go`）：

```go
package data

import (
    "fmt"
    "go-stock/backend/db"
    "go-stock/backend/logger"
    "go-stock/backend/models"
    "time"

    "github.com/go-resty/resty/v2"
    "github.com/duke-git/lancet/v2/convertor"
    "github.com/duke-git/lancet/v2/strutil"
)

// 新数据源 API 结构体
type NewDataApi struct {
    client *resty.Client
}

// 创建新数据源 API 实例
func NewNewDataApi() *NewDataApi {
    return &NewDataApi{
        client: resty.New().SetTimeout(10 * time.Second),
    }
}

// 从新数据源获取数据
func (api *NewDataApi) GetNewData(params map[string]string) (*[]models.NewDataType, error) {
    // 1. 构建 API 请求
    url := "https://api.newdatasource.com/v1/data"
    
    // 2. 发送 HTTP 请求
    res := map[string]interface{}{}
    resp, err := api.client.R().
        SetHeader("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/117.0.0.0 Safari/537.36").
        SetHeader("Content-Type", "application/json").
        SetQueryParams(params).
        SetResult(&res).
        Get(url)
    
    if err != nil {
        logger.SugaredLogger.Errorf("请求新数据源失败: %v", err)
        return nil, fmt.Errorf("请求新数据源失败: %v", err)
    }
    
    // 3. 解析响应数据
    var dataList []models.NewDataType
    
    // 检查响应状态
    if code, ok := res["code"].(float64); !ok || code != 0 {
        msg := "未知错误"
        if m, ok := res["message"].(string); ok {
            msg = m
        }
        logger.SugaredLogger.Errorf("新数据源返回错误: %s", msg)
        return nil, fmt.Errorf("新数据源返回错误: %s", msg)
    }
    
    // 解析数据列表
    if dataArray, ok := res["data"].([]interface{}); ok {
        for _, item := range dataArray {
            if itemMap, ok := item.(map[string]interface{}); ok {
                // 提取字段值
                code := strutil.GetOrDefault(itemMap["code"], "").(string)
                name := strutil.GetOrDefault(itemMap["name"], "").(string)
                value := convertor.ToFloat64(strutil.GetOrDefault(itemMap["value"], 0))
                unit := strutil.GetOrDefault(itemMap["unit"], "").(string)
                description := strutil.GetOrDefault(itemMap["description"], "").(string)
                
                // 构建数据模型
                data := models.NewDataType{
                    Code:        code,
                    Name:        name,
                    Value:       value,
                    Unit:        unit,
                    Source:      "新数据源",
                    DataTime:    time.Now(),
                    Description: description,
                }
                
                dataList = append(dataList, data)
                
                // 4. 存储到数据库
                go func(d models.NewDataType) {
                    var count int64
                    db.Dao.Model(&models.NewDataType{}).
                        Where("code = ? AND source = ?", d.Code, d.Source).
                        Count(&count)
                    
                    if count == 0 {
                        // 不存在则创建
                        if err := db.Dao.Create(&d).Error; err != nil {
                            logger.SugaredLogger.Errorf("创建新数据失败: %v", err)
                        }
                    } else {
                        // 存在则更新
                        if err := db.Dao.Model(&models.NewDataType{}).
                            Where("code = ? AND source = ?", d.Code, d.Source).
                            Updates(d).Error; err != nil {
                            logger.SugaredLogger.Errorf("更新新数据失败: %v", err)
                        }
                    }
                }(data)
            }
        }
    }
    
    logger.SugaredLogger.Infof("成功从新数据源获取 %d 条数据", len(dataList))
    return &dataList, nil
}
```

```go
package data

import (
    "fmt"
    "go-stock/backend/db"
    "go-stock/backend/logger"
    "go-stock/backend/models"
    "time"

    "github.com/go-resty/resty/v2"
    "github.com/duke-git/lancet/v2/convertor"
    "github.com/duke-git/lancet/v2/strutil"
)

// 财经数据 API
type FinancialDataApi struct {
    client *resty.Client
}

// 创建财经数据 API 实例
func NewFinancialDataApi() *FinancialDataApi {
    return &FinancialDataApi{
        client: resty.New().SetTimeout(15 * time.Second),
    }
}

// 获取财经指标数据
func (api *FinancialDataApi) GetFinancialIndicators(period string) (*[]models.FinancialIndicator, error) {
    // 1. 构建 API 请求
    url := "https://api.financialdata.com/v1/indicators"
    
    // 2. 发送 HTTP 请求
    res := map[string]interface{}{}
    resp, err := api.client.R().
        SetHeader("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/117.0.0.0 Safari/537.36").
        SetQueryParam("period", period).
        SetResult(&res).
        Get(url)
    
    if err != nil {
        logger.SugaredLogger.Errorf("请求财经数据失败: %v", err)
        return nil, fmt.Errorf("请求财经数据失败: %v", err)
    }
    
    // 3. 解析响应数据
    var indicators []models.FinancialIndicator
    
    // 检查响应状态
    if status, ok := res["status"].(string); !ok || status != "success" {
        msg := "未知错误"
        if m, ok := res["message"].(string); ok {
            msg = m
        }
        logger.SugaredLogger.Errorf("财经数据源返回错误: %s", msg)
        return nil, fmt.Errorf("财经数据源返回错误: %s", msg)
    }
    
    // 解析指标列表
    if dataArray, ok := res["data"].([]interface{}); ok {
        for _, item := range dataArray {
            if itemMap, ok := item.(map[string]interface{}); ok {
                // 提取字段
                indicatorCode := strutil.GetOrDefault(itemMap["code"], "").(string)
                indicatorName := strutil.GetOrDefault(itemMap["name"], "").(string)
                value := convertor.ToFloat64(strutil.GetOrDefault(itemMap["value"], 0))
                unit := strutil.GetOrDefault(itemMap["unit"], "").(string)
                description := strutil.GetOrDefault(itemMap["description"], "").(string)
                releaseDateStr := strutil.GetOrDefault(itemMap["releaseDate"], "").(string)
                
                // 解析发布日期
                releaseDate := time.Now()
                if releaseDateStr != "" {
                    if t, err := time.Parse("2006-01-02", releaseDateStr); err == nil {
                        releaseDate = t
                    }
                }
                
                // 构建指标模型
                indicator := models.FinancialIndicator{
                    IndicatorCode:  indicatorCode,
                    IndicatorName:  indicatorName,
                    Value:          value,
                    Unit:           unit,
                    Period:         period,
                    Source:         "财经数据源",
                    ReleaseDate:    releaseDate,
                    Description:    description,
                }
                
                indicators = append(indicators, indicator)
                
                // 4. 存储到数据库
                go func(ind models.FinancialIndicator) {
                    var count int64
                    db.Dao.Model(&models.FinancialIndicator{}).
                        Where("indicator_code = ? AND period = ? AND source = ?", 
                            ind.IndicatorCode, ind.Period, ind.Source).
                        Count(&count)
                    
                    if count == 0 {
                        if err := db.Dao.Create(&ind).Error; err != nil {
                            logger.SugaredLogger.Errorf("创建财经指标失败: %v", err)
                        }
                    } else {
                        if err := db.Dao.Model(&models.FinancialIndicator{}).
                            Where("indicator_code = ? AND period = ? AND source = ?", 
                                ind.IndicatorCode, ind.Period, ind.Source).
                            Updates(ind).Error; err != nil {
                            logger.SugaredLogger.Errorf("更新财经指标失败: %v", err)
                        }
                    }
                }(indicator)
            }
        }
    }
    
    logger.SugaredLogger.Infof("成功获取 %d 条财经指标数据", len(indicators))
    return &indicators, nil
}
```
